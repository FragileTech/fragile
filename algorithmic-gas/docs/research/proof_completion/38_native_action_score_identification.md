\tag{NASI.3}

$$
\begin{gathered}

       =\frac2{R^4}+\frac{\beta^2}{R^2}C_2(n)
                       +\beta^4 C_4(n),\\

\end{gathered}
\tag{NASI.4}

\qquad

\tag{NASI.5}
$$
